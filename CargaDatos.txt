// =====================================================================
// PRACTICA FINAL - BASES DE DATOS 2 - NEO4J
// Plataforma de cursos en linea (tipo Coursera / edX)
// Script de carga: restricciones (CONSTRAINT), nodos y relaciones
// =====================================================================
//
// Nodos (7): Usuario, Curso, Modulo, Leccion, Categoria, Idioma, Pais
// Relaciones (9): VIO, LE_GUSTA, INSCRITO_EN, RESIDE_EN, ESTA_COMPUESTO,
//                 CONTIENE, EN_IDIOMA, PERTENECE_A, DISPONIBLE_EN
//
// Orden de ejecucion: 1) Restricciones -> 2) Nodos -> 3) Relaciones
// Nota: los identificadores (U01, C01, M01, L01, CAT01, ES, CO...) son
// texto en los CSV, por eso se cargan como STRING (sin toInteger).
// =====================================================================


// =====================================================================
// 1. RESTRICCIONES (CONSTRAINT)
// =====================================================================

CREATE CONSTRAINT usuario_id IF NOT EXISTS
FOR (u:Usuario) REQUIRE u.usuarioId IS UNIQUE;

CREATE CONSTRAINT curso_id IF NOT EXISTS
FOR (c:Curso) REQUIRE c.cursoId IS UNIQUE;

CREATE CONSTRAINT modulo_id IF NOT EXISTS
FOR (m:Modulo) REQUIRE m.moduloId IS UNIQUE;

CREATE CONSTRAINT leccion_id IF NOT EXISTS
FOR (l:Leccion) REQUIRE l.leccionId IS UNIQUE;

CREATE CONSTRAINT categoria_codigo IF NOT EXISTS
FOR (cat:Categoria) REQUIRE cat.codigo IS UNIQUE;

CREATE CONSTRAINT idioma_codigo IF NOT EXISTS
FOR (i:Idioma) REQUIRE i.codigo IS UNIQUE;

CREATE CONSTRAINT pais_codigo IF NOT EXISTS
FOR (p:Pais) REQUIRE p.codigo IS UNIQUE;


// =====================================================================
// 2. CARGA DE NODOS
//    Se hace MERGE sobre el identificador unico y SET del resto de
//    propiedades: evita duplicados y permite propiedades opcionales
//    vacias (un MERGE con una propiedad null genera error).
// =====================================================================

// Nodo Usuario
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/juanbermudezm-UAO/Practica-Final/refs/heads/main/usuarios.csv' AS row
MERGE (u:Usuario {usuarioId: row.usuarioId})
SET u.nombre          = row.nombre,
    u.edad            = toInteger(row.edad),
    u.sexo            = CASE
                          WHEN trim(coalesce(row.sexo, '')) = '' THEN null
                          ELSE trim(row.sexo)
                        END,
    u.email           = row.email,
    u.fechaNacimiento = date(row.fechaNacimiento);

// Nodo Curso
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/juanbermudezm-UAO/Practica-Final/refs/heads/main/cursos.csv' AS row
MERGE (c:Curso {cursoId: row.cursoId})
SET c.titulo               = row.titulo,
    c.descripcion          = row.descripcion,
    c.nivel                = row.nivel,
    c.anioPublicacion      = toInteger(row.anioPublicacion),
    c.calificacionPromedio = toFloat(row.calificacionPromedio);

// Nodo Modulo
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/juanbermudezm-UAO/Practica-Final/refs/heads/main/modulos.csv' AS row
MERGE (m:Modulo {moduloId: row.moduloId})
SET m.titulo = row.titulo,
    m.numero = toInteger(row.numero);

// Nodo Leccion
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/juanbermudezm-UAO/Practica-Final/refs/heads/main/lecciones.csv' AS row
MERGE (l:Leccion {leccionId: row.leccionId})
SET l.titulo      = row.titulo,
    l.descripcion = row.descripcion,
    l.duracionMin = toInteger(row.duracionMin),
    l.numero      = toInteger(row.numero);

// Nodo Categoria
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/juanbermudezm-UAO/Practica-Final/refs/heads/main/categorias.csv' AS row
MERGE (cat:Categoria {codigo: row.codigo})
SET cat.nombre      = row.nombre,
    cat.descripcion = row.descripcion;

// Nodo Idioma
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/juanbermudezm-UAO/Practica-Final/refs/heads/main/idiomas.csv' AS row
MERGE (i:Idioma {codigo: row.codigo})
SET i.nombre = row.nombre;

// Nodo Pais
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/juanbermudezm-UAO/Practica-Final/refs/heads/main/paises.csv' AS row
MERGE (p:Pais {codigo: row.codigo})
SET p.nombre = row.nombre;


// =====================================================================
// 3. CARGA DE RELACIONES
// =====================================================================

// Relacion (:Usuario)-[:VIO]->(:Leccion)
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/juanbermudezm-UAO/Practica-Final/refs/heads/main/VIO.csv' AS row
MATCH (user:Usuario {usuarioId: row.usuarioId})
MATCH (leccion:Leccion {leccionId: row.leccionId})
MERGE (user)-[r:VIO {fecha: date(row.fecha)}]->(leccion)
SET r.minutosReproducidos = toInteger(row.minutosReproducidos),
    r.finalizado          = toBoolean(trim(row.finalizado));

// Relacion (:Usuario)-[:LE_GUSTA]->(:Curso)
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/juanbermudezm-UAO/Practica-Final/refs/heads/main/LE_GUSTA.csv' AS row
MATCH (user:Usuario {usuarioId: row.usuarioId})
MATCH (curso:Curso {cursoId: row.cursoId})
MERGE (user)-[r:LE_GUSTA]->(curso)
SET r.valoracion = toInteger(row.valoracion),
    r.fecha      = date(row.fecha);

// Relacion (:Usuario)-[:INSCRITO_EN]->(:Curso)
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/juanbermudezm-UAO/Practica-Final/refs/heads/main/INSCRITO_EN.csv' AS row
MATCH (user:Usuario {usuarioId: row.usuarioId})
MATCH (curso:Curso {cursoId: row.cursoId})
MERGE (user)-[r:INSCRITO_EN {fechaInicio: date(row.fechaInicio)}]->(curso)
SET r.fechaFin = CASE
        WHEN trim(coalesce(row.fechaFin, '')) = '' THEN null
        ELSE date(trim(row.fechaFin))
    END,
    r.progreso = toInteger(row.progreso),
    r.estado   = row.estado;

// Relacion (:Usuario)-[:RESIDE_EN]->(:Pais)
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/juanbermudezm-UAO/Practica-Final/refs/heads/main/RESIDE_EN.csv' AS row
MATCH (user:Usuario {usuarioId: row.usuarioId})
MATCH (pais:Pais {codigo: row.codigoPais})
MERGE (user)-[:RESIDE_EN]->(pais);

// Relacion (:Curso)-[:ESTA_COMPUESTO]->(:Modulo)
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/juanbermudezm-UAO/Practica-Final/refs/heads/main/ESTA_COMPUESTO.csv' AS row
MATCH (curso:Curso {cursoId: row.cursoId})
MATCH (modulo:Modulo {moduloId: row.moduloId})
MERGE (curso)-[r:ESTA_COMPUESTO]->(modulo)
SET r.orden = toInteger(row.orden);

// Relacion (:Modulo)-[:CONTIENE]->(:Leccion)
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/juanbermudezm-UAO/Practica-Final/refs/heads/main/CONTIENE.csv' AS row
MATCH (modulo:Modulo {moduloId: row.moduloId})
MATCH (leccion:Leccion {leccionId: row.leccionId})
MERGE (modulo)-[r:CONTIENE]->(leccion)
SET r.orden = toInteger(row.orden);

// Relacion (:Curso)-[:EN_IDIOMA]->(:Idioma)
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/juanbermudezm-UAO/Practica-Final/refs/heads/main/EN_IDIOMA.csv' AS row
MATCH (curso:Curso {cursoId: row.cursoId})
MATCH (idioma:Idioma {codigo: row.codigoIdioma})
MERGE (curso)-[:EN_IDIOMA]->(idioma);

// Relacion (:Curso)-[:PERTENECE_A]->(:Categoria)
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/juanbermudezm-UAO/Practica-Final/refs/heads/main/PERTENECE_A.csv' AS row
MATCH (curso:Curso {cursoId: row.cursoId})
MATCH (categoria:Categoria {codigo: row.codigoCategoria})
MERGE (curso)-[:PERTENECE_A]->(categoria);

// Relacion (:Curso)-[:DISPONIBLE_EN]->(:Pais)
LOAD CSV WITH HEADERS FROM 'https://raw.githubusercontent.com/juanbermudezm-UAO/Practica-Final/refs/heads/main/DISPONIBLE_EN.csv' AS row
MATCH (curso:Curso {cursoId: row.cursoId})
MATCH (pais:Pais {codigo: row.codigoPais})
MERGE (curso)-[r:DISPONIBLE_EN {fechaInicio: date(row.fechaInicio)}]->(pais)
SET r.fechaFin = CASE
    WHEN trim(coalesce(row.fechaFin, '')) = '' THEN null
    ELSE date(trim(row.fechaFin))
END;
